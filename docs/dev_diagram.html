SAM 주요 명령어

# 빌드
sam build --use-container
# 최초 배포 (guided)
sam deploy --guided --stack-name coupa-chatbot --region us-east-1
# 이후 배포
sam deploy
# 임베딩 갱신 (Lambda 2 수동 실행)
aws lambda invoke --function-name coupa-chatbot-embed --payload '{}' response.json
# 로그 확인
sam logs -n CoupaChatbotRag --tail
